const express = require("express");
const mongoose = require("mongoose");
const { getAuth } = require("firebase-admin/auth");

const router = express.Router();

const Post = require("../PublicSpacePost");
const Friendship = require("../PublicSpaceFriendship");
const requireFirebaseAuth = require("../../middleware/requireFirebaseAuth");
const adminApp = require("../../firebaseAdmin");

const firebaseAuth = getAuth(adminApp);

const EDIT_WINDOW_MINUTES = Math.max(
  1,
  Number(process.env.PUBLIC_SPACE_EDIT_MINUTES) || 15
);

const DUPLICATE_WINDOW_MINUTES = 10;
const MAX_TEXT_LENGTH = 5000;
const MAX_MEDIA_ITEMS = 4;

function getDailyPostLimit(friendCount) {
  if (friendCount === 0) return 0;
  if (friendCount === 1) return 1;
  if (friendCount <= 5) return 2;
  if (friendCount <= 10) return 5;
  return Infinity;
}

function getMediaType(text, media) {
  if (media.length === 0) return "text";
  if (text && media.length > 0) return "mixed";
  return media[0].type;
}

function getUtcDayStart() {
  const now = new Date();

  return new Date(
    Date.UTC(now.getUTCFullYear(), now.getUTCMonth(), now.getUTCDate())
  );
}

function extractHashtags(text) {
  return [
    ...new Set(
      (text.match(/#[\p{L}\p{N}_]+/gu) || []).map((tag) =>
        tag.slice(1).toLowerCase()
      )
    ),
  ].slice(0, 50);
}

function extractMentions(text) {
  return [
    ...new Set(
      (text.match(/@[\p{L}\p{N}_]+/gu) || []).map((mention) =>
        mention.slice(1).toLowerCase()
      )
    ),
  ].slice(0, 50);
}

// Firebase user records provide displayName and photoURL.
// Firebase does not have a native username field, so username is
// generated from the display name as a fallback.

async function getAuthorProfile(uid) {
  try {
    const firebaseUser = await firebaseAuth.getUser(uid);

    const displayName = firebaseUser.displayName || "";
    const username = displayName
      ? displayName
          .normalize("NFKD")
          .replace(/[^\p{L}\p{N}_]+/gu, "")
          .toLowerCase()
          .slice(0, 30)
      : "";

    console.log("Public Space author profile:", {
      uid: firebaseUser.uid,
      displayName,
      hasPhoto: Boolean(firebaseUser.photoURL),
    });

    return {
      displayName,
      username,
      photoURL: firebaseUser.photoURL || "",
    };
  } catch (error) {
    console.error("Could not load Public Space author profile:", {
      code: error.code,
      message: error.message,
      uid,
    });

    return {
      displayName: "",
      username: "",
      photoURL: "",
    };
  }
}


async function getAcceptedFriendCount(uid) {
  return Friendship.countDocuments({
    status: "accepted",
    $or: [{ userAUid: uid }, { userBUid: uid }],
  });
}

function validateMedia(rawMedia) {
  if (!Array.isArray(rawMedia) || rawMedia.length > MAX_MEDIA_ITEMS) {
    return {
      error: `You can attach up to ${MAX_MEDIA_ITEMS} media items.`,
    };
  }

  const media = [];

  for (const item of rawMedia) {
    if (
      !item ||
      !["image", "video"].includes(item.type) ||
      typeof item.url !== "string"
    ) {
      return { error: "Each media item needs a valid type and URL." };
    }

    let parsedUrl;

    try {
      parsedUrl = new URL(item.url);
    } catch {
      return { error: "A media URL is invalid." };
    }

    if (parsedUrl.protocol !== "https:") {
      return { error: "Media URLs must use HTTPS." };
    }

    media.push({
      type: item.type,
      url: parsedUrl.toString(),
    });
  }

  return { media };
}

function mediaSignature(media = []) {
  return media
    .map((item) => `${item.type}:${item.url}`)
    .sort()
    .join("|");
}

// Create a post.
router.post("/", requireFirebaseAuth, requireVerifiedLogin, async (req, res) => {
  try {
    const uid = req.firebaseUser.uid;

    const text =
      typeof req.body.text === "string" ? req.body.text.trim() : "";

    const privacy = req.body.privacy || "public";
    const rawMedia = req.body.media ?? [];

    if (!["public", "friends"].includes(privacy)) {
      return res.status(400).json({
        success: false,
        message: "Privacy must be public or friends.",
      });
    }

    if (text.length > MAX_TEXT_LENGTH) {
      return res.status(400).json({
        success: false,
        message: `Post text cannot exceed ${MAX_TEXT_LENGTH} characters.`,
      });
    }

    const mediaResult = validateMedia(rawMedia);

    if (mediaResult.error) {
      return res.status(400).json({
        success: false,
        message: mediaResult.error,
      });
    }

    const media = mediaResult.media;

    if (!text && media.length === 0) {
      return res.status(400).json({
        success: false,
        message: "Add text, an image, or a video to create a post.",
      });
    }

    const friendCount = await getAcceptedFriendCount(uid);
    const dailyLimit = getDailyPostLimit(friendCount);

    if (dailyLimit === 0) {
      return res.status(403).json({
        success: false,
        message: "You need at least one accepted friend to publish posts.",
        friendCount,
        dailyLimit: 0,
      });
    }

    const dayStart = getUtcDayStart();

    if (Number.isFinite(dailyLimit)) {
      const postsToday = await Post.countDocuments({
        authorUid: uid,
        isDeleted: false,
        createdAt: { $gte: dayStart },
      });

      if (postsToday >= dailyLimit) {
        return res.status(429).json({
          success: false,
          message: "You have reached your daily posting limit.",
          friendCount,
          dailyLimit,
          postsToday,
        });
      }
    }

    const duplicateSince = new Date(
      Date.now() - DUPLICATE_WINDOW_MINUTES * 60 * 1000
    );

    const recentPosts = await Post.find({
      authorUid: uid,
      isDeleted: false,
      createdAt: { $gte: duplicateSince },
    })
      .select("text media")
      .lean();

    const signature = mediaSignature(media);

    const duplicate = recentPosts.some(
      (post) =>
        text === (post.text || "") &&
        signature === mediaSignature(post.media || [])
    );

    if (duplicate) {
      return res.status(409).json({
        success: false,
        message: "This looks like a duplicate of a recent post.",
      });
    }

    const authorProfile = await getAuthorProfile(uid);

    const editUntil = new Date(
      Date.now() + EDIT_WINDOW_MINUTES * 60 * 1000
    );

    const post = await Post.create({
      authorUid: uid,
      authorProfile,
      friendCountAtPosting: friendCount,
      text,
      media,
      mediaType: getMediaType(text, media),
      privacy,
      hashtags: extractHashtags(text),
      mentions: extractMentions(text),
      editUntil,
    });

    return res.status(201).json({
      success: true,
      message: "Post created successfully.",
      post,
      friendCount,
      dailyLimit: Number.isFinite(dailyLimit) ? dailyLimit : null,
    });
  } catch (error) {
    console.error("Create Public Space post error:", error.message);

    return res.status(500).json({
      success: false,
      message: "Could not create post.",
    });
  }
});

// View recent public posts.
router.get("/", requireFirebaseAuth, requireVerifiedLogin, async (req, res) => {
  try {
    const posts = await Post.find({
      privacy: "public",
      isDeleted: false,
    })
      .sort({ createdAt: -1 })
      .limit(20)
      .lean();

    // Refresh author profiles from Firebase for older posts too.
    const profileCache = new Map();

    const postsWithProfiles = await Promise.all(
      posts.map(async (post) => {
        if (!profileCache.has(post.authorUid)) {
          profileCache.set(
            post.authorUid,
            getAuthorProfile(post.authorUid)
          );
        }

        const authorProfile = await profileCache.get(post.authorUid);

        return {
          ...post,
          authorProfile,
        };
      })
    );

    return res.json({
      success: true,
      count: postsWithProfiles.length,
      posts: postsWithProfiles,
    });
  } catch (error) {
    console.error("Load Public Space posts error:", error.message);

    return res.status(500).json({
      success: false,
      message: "Could not load posts.",
    });
  }
});

// Edit an existing post.
router.patch("/:postId", requireFirebaseAuth, requireVerifiedLogin, async (req, res) => {
  try {
    const uid = req.firebaseUser.uid;
    const { postId } = req.params;

    if (!mongoose.Types.ObjectId.isValid(postId)) {
      return res.status(400).json({
        success: false,
        message: "Invalid post ID.",
      });
    }

    const post = await Post.findOne({
      _id: postId,
      authorUid: uid,
      isDeleted: false,
    });

    if (!post) {
      return res.status(404).json({
        success: false,
        message: "Post not found or you are not its author.",
      });
    }

    if (!post.editUntil || new Date() > post.editUntil) {
      return res.status(403).json({
        success: false,
        message: "The editing time limit has expired.",
      });
    }

    const { text, media, privacy } = req.body;

    if (typeof text !== "string" || !Array.isArray(media)) {
      return res.status(400).json({
        success: false,
        message: "Provide text and a media array.",
      });
    }

    const cleanText = text.trim();

    if (cleanText.length > MAX_TEXT_LENGTH) {
      return res.status(400).json({
        success: false,
        message: `Post text cannot exceed ${MAX_TEXT_LENGTH} characters.`,
      });
    }

    const mediaResult = validateMedia(media);

    if (mediaResult.error) {
      return res.status(400).json({
        success: false,
        message: mediaResult.error,
      });
    }

    const cleanMedia = mediaResult.media;

    if (!cleanText && cleanMedia.length === 0) {
      return res.status(400).json({
        success: false,
        message: "A post must contain text, an image, or a video.",
      });
    }

    if (
      privacy !== undefined &&
      !["public", "friends"].includes(privacy)
    ) {
      return res.status(400).json({
        success: false,
        message: "Privacy must be public or friends.",
      });
    }

    post.text = cleanText;
    post.media = cleanMedia;
    post.mediaType = getMediaType(cleanText, cleanMedia);
    post.hashtags = extractHashtags(cleanText);
    post.mentions = extractMentions(cleanText);

    if (privacy !== undefined) {
      post.privacy = privacy;
    }

    await post.save();

    return res.json({
      success: true,
      message: "Post updated successfully.",
      post,
    });
  } catch (error) {
    console.error("Edit Public Space post error:", error.message);

    return res.status(500).json({
      success: false,
      message: "Could not update post.",
    });
  }
});

// Soft-delete a post.
router.delete("/:postId", requireFirebaseAuth, requireVerifiedLogin, async (req, res) => {
  try {
    const uid = req.firebaseUser.uid;
    const { postId } = req.params;

    if (!mongoose.Types.ObjectId.isValid(postId)) {
      return res.status(400).json({
        success: false,
        message: "Invalid post ID.",
      });
    }

    const post = await Post.findOneAndUpdate(
      {
        _id: postId,
        authorUid: uid,
        isDeleted: false,
      },
      {
        $set: {
          isDeleted: true,
          deletedAt: new Date(),
        },
      },
      { returnDocument: "after" }
    );

    if (!post) {
      return res.status(404).json({
        success: false,
        message: "Post not found or you are not its author.",
      });
    }

    return res.json({
      success: true,
      message: "Post deleted successfully.",
    });
  } catch (error) {
    console.error("Delete Public Space post error:", error.message);

    return res.status(500).json({
      success: false,
      message: "Could not delete post.",
    });
  }
});

module.exports = router;

const express = require("express");
const mongoose = require("mongoose");
const { getAuth } = require("firebase-admin/auth");

const { adminApp } = require("../../firebaseAdmin");
const Friendship = require("../PublicSpaceFriendship");
const PublicSpaceNotification = require("../PublicSpaceNotification");

const router = express.Router();
const auth = getAuth(adminApp);

const asyncHandler = (handler) => (req, res, next) => {
  Promise.resolve(handler(req, res, next)).catch(next);
};

const requireFirebaseAuth = asyncHandler(async (req, res, next) => {
  const header = req.headers.authorization || "";
  const token = header.startsWith("Bearer ")
    ? header.slice(7)
    : "";

  if (!token) {
    return res.status(401).json({
      success: false,
      message: "Authentication required.",
    });
  }

  try {
    const decoded = await auth.verifyIdToken(token);
    req.user = { uid: decoded.uid };
    next();
  } catch {
    return res.status(401).json({
      success: false,
      message: "Invalid or expired login.",
    });
  }
});

router.use(requireFirebaseAuth);
router.use(requireVerifiedLogin);

async function createNotification({
  recipientUid,
  actorUid,
  type,
  message,
  friendshipId,
}) {
  if (!recipientUid || !actorUid || recipientUid === actorUid) return;

  try {
    await PublicSpaceNotification.create({
      recipientUid,
      actorUid,
      type,
      message,
      friendshipId,
    });
  } catch (error) {
    // Notification failures should not cancel a successful friend action.
    console.error("Friend notification creation failed:", error.message);
  }
}

function getPair(uid1, uid2) {
  return [uid1, uid2].sort();
}

// SEND FRIEND REQUEST
router.post(
  "/request/:targetUid",
  asyncHandler(async (req, res) => {
    const uid = req.user.uid;
    const targetUid = String(req.params.targetUid || "").trim();

    if (!targetUid || targetUid === uid) {
      return res.status(400).json({
        message: "You cannot send a friend request to yourself.",
      });
    }

    // Confirm that the target is a valid Firebase user.
    try {
      await auth.getUser(targetUid);
    } catch {
      return res.status(404).json({
        message: "The target user was not found.",
      });
    }

    const [userAUid, userBUid] = getPair(uid, targetUid);

    let friendship = await Friendship.findOne({
      userAUid,
      userBUid,
    });

    if (friendship) {
      if (friendship.status === "accepted") {
        return res.status(409).json({
          message: "You are already friends.",
          friendship,
        });
      }

      if (friendship.status === "pending") {
        return res.status(409).json({
          message: "A friend request is already pending.",
          friendship,
        });
      }

      if (friendship.status === "blocked") {
        return res.status(403).json({
          message: "This friendship request cannot be sent.",
        });
      }

      // Permit a new request after a previous request was rejected.
      friendship.requestedByUid = uid;
      friendship.status = "pending";
      friendship.respondedAt = null;
      friendship.blockedByUid = null;
      await friendship.save();
    } else {
      friendship = await Friendship.create({
        userAUid,
        userBUid,
        requestedByUid: uid,
        status: "pending",
      });
    }

    await createNotification({
      recipientUid: targetUid,
      actorUid: uid,
      type: "friend_request",
      message: "You received a new friend request.",
      friendshipId: friendship._id,
    });

    return res.status(201).json({
      message: "Friend request sent.",
      friendship,
    });
  })
);

// ACCEPT OR REJECT AN INCOMING REQUEST
router.patch(
  "/:friendshipId/respond",
  asyncHandler(async (req, res) => {
    const uid = req.user.uid;
    const { friendshipId } = req.params;
    const action =
      typeof req.body.action === "string"
        ? req.body.action.trim().toLowerCase()
        : "";

    if (!mongoose.Types.ObjectId.isValid(friendshipId)) {
      return res.status(400).json({
        message: "Invalid friendship ID.",
      });
    }

    if (!["accept", "reject"].includes(action)) {
      return res.status(400).json({
        message: "Action must be accept or reject.",
      });
    }

    const friendship = await Friendship.findOne({
      _id: friendshipId,
      status: "pending",
    });

    if (!friendship) {
      return res.status(404).json({
        message: "Pending friend request not found.",
      });
    }

    if (
      friendship.requestedByUid === uid ||
      (friendship.userAUid !== uid && friendship.userBUid !== uid)
    ) {
      return res.status(403).json({
        message: "You cannot respond to this friend request.",
      });
    }

    friendship.status = action === "accept" ? "accepted" : "rejected";
    friendship.respondedAt = new Date();
    await friendship.save();

    if (action === "accept") {
      await createNotification({
        recipientUid: friendship.requestedByUid,
        actorUid: uid,
        type: "friend_accepted",
        message: "Your friend request was accepted.",
        friendshipId: friendship._id,
      });
    }

    return res.json({
      message:
        action === "accept"
          ? "Friend request accepted."
          : "Friend request rejected.",
      friendship,
    });
  })
);

    const friends = friendships.map((friendship) => ({

    return res.json({ friends });
  })
);

const mongoose = require("mongoose");

const firebaseAuth = getAuth(adminApp);

const asyncHandler = (handler) => (req, res, next) => {

// Verify the Firebase ID token.
const requireFirebaseAuth = asyncHandler(async (req, res, next) => {
  const header = req.headers.authorization || "";
  const token = header.startsWith("Bearer ")
    ? header.slice(7)
    : "";

  if (!token) {
    return res.status(401).json({

  try {
    const decoded = await firebaseAuth.verifyIdToken(token);

const validId = (id) => mongoose.Types.ObjectId.isValid(id);

// Create a notification without allowing a notification failure
// to turn a successful like/comment into a failed request.
async function createNotification({
  recipientUid,
  actorUid,
  type,
  message,
  postId = null,
}) {
  if (!recipientUid || !actorUid || recipientUid === actorUid) {
    return;
  }

  try {
    // Get the actor's real name from Firebase.
    const actor = await firebaseAuth.getUser(actorUid);

    const actorName =
      (typeof actor.displayName === "string" &&
        actor.displayName.trim()) ||
      (typeof actor.email === "string" &&
        actor.email.includes("@") &&
        actor.email.split("@")[0]) ||
      "Someone";

    // Replace the generic "Someone" with the actor's name.
    const personalizedMessage = message.replace(
      /^Someone\b/,
      actorName
    );

      await PublicSpaceNotification.create({
      recipientUid,
      actorUid,
      type,
      message: personalizedMessage,
      postId,
    });
  } catch (error) {
    console.error("Notification creation failed:", error.message);
  }
}

async function findAvailablePost(postId, uid, res) {
  if (!validId(postId)) {
    res.status(400).json({ message: "Invalid post ID." });
    return null;
  }

  const post = await PublicSpacePost.findOne({
    _id: postId,
    isDeleted: false,
  });

  if (!post) {
    res.status(404).json({ message: "Post not found." });
    return null;
  }

  // Public posts and the current user's own posts are available.
  // Friends-only access will be added in the feed/privacy upgrade.
  if (post.privacy !== "public" && post.authorUid !== uid) {
    res.status(403).json({
      message: "You cannot interact with this private post.",
    });
    return null;
  }

  return post;
}

// GET MY INTERACTION STATUS
router.get(
  "/status",
  asyncHandler(async (req, res) => {
    const rawPostIds =
      typeof req.query.postIds === "string"
        ? req.query.postIds
        : "";

    const postIds = [
      ...new Set(
        rawPostIds
          .split(",")
          .map((id) => id.trim())
          .filter(Boolean)
      ),
    ];

    if (postIds.length > 100) {
      return res.status(400).json({
        message: "Maximum 100 post IDs per request.",
      });
    }

    if (postIds.some((id) => !validId(id))) {
      return res.status(400).json({
        message: "One or more post IDs are invalid.",
      });
    }

    if (postIds.length === 0) {
      return res.json({
        likedPostIds: [],
        savedPostIds: [],
      });
    }

    const interactions = await PublicSpaceInteraction.find({
      userUid: req.user.uid,
      postId: { $in: postIds },
      type: { $in: ["like", "save"] },
    })
      .select("postId type")
      .lean();

    const likedPostIds = interactions
      .filter((item) => item.type === "like")
      .map((item) => String(item.postId));

    const savedPostIds = interactions
      .filter((item) => item.type === "save")
      .map((item) => String(item.postId));

    res.json({
      likedPostIds,
      savedPostIds,
    });
  })
);

// LIKE / UNLIKE
router.post(
  "/:postId/like",
  asyncHandler(async (req, res) => {
    const uid = req.user.uid;
    const post = await findAvailablePost(req.params.postId, uid, res);
    if (!post) return;

    const filter = {
      postId: post._id,
      userUid: uid,
      type: "like",
    };

    const existing = await PublicSpaceInteraction.findOne(filter);
    let liked;

    if (existing) {
      await PublicSpaceInteraction.deleteOne({ _id: existing._id });
      liked = false;
    } else {
      try {
        await PublicSpaceInteraction.create(filter);
        liked = true;
      } catch (error) {
        if (error.code !== 11000) throw error;

        // Another request may have created the like concurrently.
        liked = Boolean(
          await PublicSpaceInteraction.exists(filter)

      }

        await createNotification({
          recipientUid: post.authorUid,
          actorUid: uid,
          type: "like",
          message: "Someone liked your post.",
          postId: post._id,
        });
      }
    }

    const likesCount = await PublicSpaceInteraction.countDocuments({
      postId: post._id,
      type: "like",
    });

    await PublicSpacePost.updateOne(
      { _id: post._id },
      { $set: { likesCount } }
    );

    res.json({ liked, likesCount });
  })
);

  "/:postId/report",
  asyncHandler(async (req, res) => {
    const uid = req.user.uid;
    const post = await findAvailablePost(req.params.postId, uid, res);
    if (!post) return;

    if (post.authorUid === uid) {
      return res.status(400).json({
        message: "You cannot report your own post.",
      });
    }

    const allowedReasons = [
      "spam",
      "harassment",
      "inappropriate",
      "misinformation",
      "other",
    ];

    const reason =
      typeof req.body.reason === "string"
        ? req.body.reason.trim().toLowerCase()
        : "";

    if (!allowedReasons.includes(reason)) {
      return res.status(400).json({
        message: "Choose a valid report reason.",
        allowedReasons,
      });
    }

    try {
      await PublicSpaceInteraction.create({
        postId: post._id,
        userUid: uid,
        type: "report",
        reportReason: reason,
      });
    } catch (error) {
      if (error.code === 11000) {
        return res.status(409).json({
          message: "You have already reported this post.",
        });
      }
      throw error;
    }

    res.status(201).json({
      message: "Report submitted for review.",
    });
  })
);

  Promise.resolve(handler(req, res, next)).catch(next);

const requireFirebaseAuth = asyncHandler(async (req, res, next) => {
  const header = req.headers.authorization || "";
  const token = header.startsWith("Bearer ")
    ? header.slice(7)
    : "";

  if (!token) {
    return res.status(401).json({
      success: false,
      message: "Authentication required.",
    });
  }

  try {
    const decoded = await firebaseAuth.verifyIdToken(token);
    req.user = { uid: decoded.uid };
    next();
  } catch {
    return res.status(401).json({
      success: false,
      message: "Invalid or expired login.",
    });
  }
});

const express = require("express");
const multer = require("multer");
const cloudinary = require("cloudinary").v2;

const router = express.Router();
const requireFirebaseAuth = require("../../middleware/requireFirebaseAuth");

const MAX_VIDEO_SIZE = 50 * 1024 * 1024; // 50 MB

const ALLOWED_VIDEO_TYPES = [
  "video/mp4",
  "video/webm",
  "video/quicktime",
  "video/ogg",
];

const upload = multer({
  storage: multer.memoryStorage(),
  limits: {
    fileSize: MAX_VIDEO_SIZE,
    files: 1,
  },
  fileFilter: (req, file, callback) => {
    if (!ALLOWED_VIDEO_TYPES.includes(file.mimetype)) {
      return callback(
        new Error("Unsupported video format. Please use MP4, WebM, MOV or OGG.")
      );
    }

    callback(null, true);
  },
});

function configureCloudinary() {
  const { CLOUDINARY_CLOUD_NAME, CLOUDINARY_API_KEY, CLOUDINARY_API_SECRET } =
    process.env;

  if (
    !CLOUDINARY_CLOUD_NAME ||
    !CLOUDINARY_API_KEY ||
    !CLOUDINARY_API_SECRET
  ) {
    throw new Error("Cloudinary environment variables are not configured.");
  }

  cloudinary.config({
    cloud_name: CLOUDINARY_CLOUD_NAME,
    api_key: CLOUDINARY_API_KEY,
    api_secret: CLOUDINARY_API_SECRET,
    secure: true,
  });
}

router.post(
  "/video",
  requireFirebaseAuth,
  (req, res, next) => {
    upload.single("video")(req, res, (error) => {
      if (error) {
        const isTooLarge = error instanceof multer.MulterError &&
          error.code === "LIMIT_FILE_SIZE";

        return res.status(isTooLarge ? 413 : 400).json({
          message: isTooLarge
            ? "Video exceeds the 50 MB upload limit."
            : error.message || "Unable to receive video.",
        });
      }

      next();
    });
  },
  async (req, res) => {
    try {
      if (!req.file) {
        return res.status(400).json({
          message: "Please select a video to upload.",
        });
      }

      configureCloudinary();

      const uploaded = await new Promise((resolve, reject) => {
        const stream = cloudinary.uploader.upload_stream(
          {
            resource_type: "video",
            folder: "internarea/public-space/videos",
          },
          (error, result) => {
            if (error) return reject(error);
            resolve(result);
          }
        );

        stream.end(req.file.buffer);
      });

      return res.status(201).json({
        message: "Video uploaded successfully.",
        media: {
          type: "video",
          url: uploaded.secure_url,
          storageKey: uploaded.public_id,
          contentHash: uploaded.etag || "",
        },
      });
    } catch (error) {
      console.error("Public Space video upload failed:", error);

      return res.status(500).json({
        message: "Video upload failed. Please try again.",
      });
    }
  }
);

module.exports = router;
